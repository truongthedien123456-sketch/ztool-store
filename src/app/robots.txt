import { MetadataRoute } from 'next';

export default function robots(): MetadataRoute.Robots {
  return {
    rules: {
      userAgent: '*',
      allow: '/',
      disallow: ['/quan-ly-secret/'], // Khóa trang Admin không cho Google hiển thị
    },
    sitemap: 'https://ztool-store.vercel.app/sitemap.xml',
  };
}
